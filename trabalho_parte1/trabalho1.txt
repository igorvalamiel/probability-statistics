import numpy as np
import matplotlib.pyplot as plt
import scipy.stats as stats

def atualizacao_bayesiana_gamma(dados, k, alpha, delta):
    
    n = len(dados)  # número de observações
    soma_dados = np.sum(dados)  # soma dos dados
    
    # Fórmulas de atualização do modelo conjugado
    alpha_post = alpha + n * k
    delta_post = delta + soma_dados
    
    return alpha_post, delta_post

def plotar_distribuicoes(alpha_priori, delta_priori, alpha_posteriori, delta_posteriori):
    """
    Plota as distribuições a priori e a posteriori no mesmo gráfico.
    """
    # Criar valores de lambda para o eixo x
    lambda_values = np.linspace(0, 3, 1000)
    
    # Calcular as densidades de probabilidade
    pdf_priori = stats.gamma.pdf(lambda_values, a=alpha_priori, scale=1/delta_priori)
    pdf_posteriori = stats.gamma.pdf(lambda_values, a=alpha_posteriori, scale=1/delta_posteriori)
    
    # Criar o gráfico
    plt.figure(figsize=(10, 6))
    
    # Plotar as distribuições
    plt.plot(lambda_values, pdf_priori, 'b-', linewidth=2, label='Priori', alpha=0.7)
    plt.plot(lambda_values, pdf_posteriori, 'r-', linewidth=2, label='Posteriori', alpha=0.7)
    
    # Adicionar linhas verticais para as médias
    media_priori = alpha_priori / delta_priori
    media_posteriori = alpha_posteriori / delta_posteriori
    
    plt.axvline(media_priori, color='blue', linestyle='--', alpha=0.7, label=f'Média Priori: {media_priori:.3f}')
    plt.axvline(media_posteriori, color='red', linestyle='--', alpha=0.7, label=f'Média Posteriori: {media_posteriori:.3f}')
    
    # Configurações do gráfico
    plt.xlabel('λ (Parâmetro de Taxa)')
    plt.ylabel('Densidade de Probabilidade')
    plt.title('Atualização Bayesiana - Modelo Gamma-Gamma')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.ylim(bottom=0)
    
    # Adicionar informações no gráfico
    plt.text(0.02, 0.98, f'Priori: Gamma(α={alpha_priori}, δ={delta_priori})', 
             transform=plt.gca().transAxes, verticalalignment='top',
             bbox=dict(boxstyle='round', facecolor='lightblue', alpha=0.8))
    
    plt.text(0.02, 0.88, f'Posteriori: Gamma(α={alpha_posteriori:.1f}, δ={delta_posteriori:.1f})', 
             transform=plt.gca().transAxes, verticalalignment='top',
             bbox=dict(boxstyle='round', facecolor='lightcoral', alpha=0.8))
    
    plt.tight_layout()
    plt.show()

# Exemplo de uso da função
if __name__ == "__main__":
    # Dados de exemplo (valores positivos, como requer a distribuição Gamma)
    np.random.seed(42)  # Para reproducibilidade
    dados_reais = np.random.gamma(shape=2, scale=0.5, size=20)  # Gerando dados de uma Gamma(2, 2)
    
    # Parâmetros fixos e hiperparâmetros
    k = 2.0        # parâmetro de forma fixo da verossimilhança
    alpha = 3.0    # hiperparâmetro de forma da priori
    delta = 2.0    # hiperparâmetro de taxa da priori
    
    print("=== ANÁLISE BAYESIANA GAMMA-GAMMA ===")
    print(f"Tamanho da amostra: n = {len(dados_reais)}")
    print(f"Soma dos dados: ∑x_i = {np.sum(dados_reais):.2f}")
    print(f"Parâmetros: k = {k}, alpha = {alpha}, delta = {delta}")
    print()
    
    # Chamando a função de atualização bayesiana
    alpha_post, delta_post = atualizacao_bayesiana_gamma(dados_reais, k, alpha, delta)
    
    print("=== RESULTADOS DA ATUALIZAÇÃO ===")
    print(f"Parâmetros da distribuição a posteriori:")
    print(f"alpha_post = {alpha_post:.2f}")
    print(f"delta_post = {delta_post:.2f}")
    print()
    
    # Calculando a média da distribuição a posteriori
    media_priori = alpha / delta
    media_posteriori = alpha_post / delta_post
    
    print("=== COMPARAÇÃO ===")
    print(f"Priori:     Gamma(shape={alpha}, rate={delta})")
    print(f"Posteriori: Gamma(shape={alpha_post:.2f}, rate={delta_post:.2f})")
    print(f"Média priori:     {media_priori:.4f}")
    print(f"Média posteriori: {media_posteriori:.4f}")
    print(f"Variação: {((media_posteriori - media_priori)/media_priori * 100):+.1f}%")
    
    # Podemos também calcular a moda (se alpha_post > 1)
    if alpha_post > 1:
        moda_posteriori = (alpha_post - 1) / delta_post
        print(f"Moda posteriori:  {moda_posteriori:.4f}")
    
    print("\nGerando gráfico...")
    
    # Gerar o gráfico
    plotar_distribuicoes(alpha, delta, alpha_post, delta_post)
    
    # Análise adicional
    print("\n=== ANÁLISE ADICIONAL ===")
    print("Intervalo de 95% de credibilidade para λ:")
    inferior_95 = stats.gamma.ppf(0.025, a=alpha_post, scale=1/delta_post)
    superior_95 = stats.gamma.ppf(0.975, a=alpha_post, scale=1/delta_post)
    print(f"λ ∈ [{inferior_95:.4f}, {superior_95:.4f}]")
